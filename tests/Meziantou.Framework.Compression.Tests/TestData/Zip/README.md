# Zip fixtures

The archives of this folder are generated by the script below, with 7-Zip 26.06 (`7zz`), Info-ZIP Zip 3.0 (`zip`),
bsdtar (libarchive, macOS), `zstd` and Python 3.9 (`zipfile`, and a small writer for what `zipfile` does not write).
They contain the same small files, whose SHA-256 the tests check (`ZipTestData.cs`):

| File | Size | Content |
| --- | --- | --- |
| `hello.txt` | 14 | `Hello, world!` and a line feed |
| `lorem.txt` | 2511 | 400 words drawn from a list with `random.Random(1)` |
| `random.bin` | 1024 | `random.Random(2).randbytes(1024)` |
| `empty.txt` | 0 | |
| `dir/nested.txt` | 140 | `nested` and a line feed, 20 times |
| `big.bin` (split archives) | 70000 | `random.Random(3).randbytes(70000)` |
| `tricky.bin` | 26 | a data descriptor signature and a header signature in stored data |

| Archive | Tool | What it covers |
| --- | --- | --- |
| `stored.zip`, `deflate.zip`, `deflate64.zip`, `bzip2.zip`, `lzma.zip`, `ppmd.zip` | `7zz a -tzip -mm=Copy\|Deflate\|Deflate64\|BZip2\|LZMA\|PPMd` | the methods, NTFS times (LZMA with an end marker) |
| `zipcrypto.zip`, `aes128.zip`, `aes192.zip`, `aes256.zip` | `7zz a -tzip -mm=Deflate -mem=... -ppassword` | ZipCrypto, WinZip AES (AE-2) |
| `aes256-stored.zip` | `7zz a -tzip -mm=Copy -mem=AES256 -ppassword` | AES around stored content |
| `split7z.zip.001`, `.002`, `.003` | `7zz a -tzip -mm=Copy -v1500b` | the parts of 7-Zip (disk numbers 0) |
| `zipcrypto-infozip.zip` | `zip -X -P password -r` | ZipCrypto with data descriptors (the check byte is the time) and sizes in the local headers |
| `streamed-infozip.zip` | `zip -X -r - ... \| cat` | streamed entries: data descriptors, sizes unknown in the local headers |
| `stdin-infozip.zip` | `zip - -` (from stdin) | an entry named `-`, zip64 local header |
| `symlink.zip` | `zip -y` | a Unix symbolic link, extended timestamps |
| `split-infozip.z01`, `split-infozip.zip` | `zip -0 -s 64k` | an Info-ZIP split archive (disk numbers, split marker) |
| `bsdtar.zip` | `bsdtar --format zip` | data descriptors |
| `aes-bsdtar.zip`, `zipcrypto-bsdtar.zip` | `bsdtar --format zip --options zip:encryption=aes256\|zipcrypt --passphrase password` | encrypted streamed entries, AE-1 and AE-2 |
| `descriptor-python.zip`, `descriptor-zip64-python.zip` | `zipfile` writing to a non-seekable stream (`force_zip64` for the second) | data descriptors, 64-bit data descriptors |
| `zip64-python.zip` | `zipfile` with `force_zip64=True` | zip64 local extra fields |
| `lzma-eos-python.zip` | `zipfile` with `ZIP_LZMA` | LZMA with an end marker |
| `comments.zip`, `utf8-names.zip`, `empty.zip` | `zipfile` | comments, UTF-8 names, an archive without entries |
| `zstd.zip`, `xz.zip` | the writer of the script | methods 93 and 95 |
| `zip64-eocd.zip` | the writer of the script | zip64 extra fields in the central directory, zip64 end of central directory |
| `descriptor-stored.zip`, `descriptor-stored-zip64.zip`, `descriptor-deflate-nosig.zip` | the writer of the script | stored streamed entries, data descriptors with and without signature |
| `cp437-names.zip` | the writer of the script | a code page 437 name with backslashes (MS-DOS), an Info-ZIP Unicode path extra field |
| `timestamps.zip` | the writer of the script | NTFS and extended timestamp extra fields, MS-DOS time |
| `sfx.zip` | the script | `deflate.zip` after 1024 bytes of data, without adjusting the offsets |

The legacy methods (Shrink, Reduce, Implode) are tested with the archives of `TestData/ZipLegacy`.

Run the script with `python3 generate.py <this folder> <path of 7zz>`:

```python
# Generates the zip fixtures of tests/Meziantou.Framework.Compression.Tests/TestData/Zip.
# Usage: python3 generate.py <output directory> <path of 7zz>
import hashlib, io, lzma, os, random, shutil, struct, subprocess, sys, tempfile, zipfile, zlib

out = os.path.abspath(sys.argv[1])
sevenzip = sys.argv[2]
os.makedirs(out, exist_ok=True)
work = tempfile.mkdtemp()

# The content of the archives
rng = random.Random(1)
words = ["lorem", "ipsum", "dolor", "sit", "amet", "consectetur", "adipiscing", "elit", "sed", "do", "eiusmod", "tempor"]
lorem = " ".join(rng.choice(words) for _ in range(400)).encode() + b"\n"
files = {
    "hello.txt": b"Hello, world!\n",
    "lorem.txt": lorem,
    "random.bin": random.Random(2).randbytes(1024),
    "empty.txt": b"",
    "dir/nested.txt": b"nested\n" * 20,
}
for name, data in files.items():
    print(name, len(data), hashlib.sha256(data).hexdigest())

src = os.path.join(work, "src")
os.makedirs(os.path.join(src, "dir"))
for name, data in files.items():
    with open(os.path.join(src, name), "wb") as f:
        f.write(data)
for root, dirs, names in os.walk(src):
    for n in dirs + names:
        os.utime(os.path.join(root, n), (1704164645, 1704164645))
os.utime(src, (1704164645, 1704164645))

def run(args, cwd=src, **kwargs):
    subprocess.run(args, cwd=cwd, check=True, stdout=subprocess.DEVNULL, **kwargs)

def target(name):
    path = os.path.join(out, name)
    if os.path.exists(path):
        os.remove(path)
    return path

inputs = ["hello.txt", "lorem.txt", "random.bin", "empty.txt", "dir"]

# 7-Zip: every method and every encryption
for method, name in [("Copy", "stored"), ("Deflate", "deflate"), ("Deflate64", "deflate64"), ("BZip2", "bzip2"), ("LZMA", "lzma"), ("PPMd", "ppmd")]:
    run([sevenzip, "a", "-tzip", "-mm=" + method, target(name + ".zip")] + inputs)
for em, name in [("ZipCrypto", "zipcrypto"), ("AES128", "aes128"), ("AES192", "aes192"), ("AES256", "aes256")]:
    run([sevenzip, "a", "-tzip", "-mm=Deflate", "-mem=" + em, "-ppassword", target(name + ".zip")] + inputs)
# AES around stored entries, and a short file (AE-2)
run([sevenzip, "a", "-tzip", "-mm=Copy", "-mem=AES256", "-ppassword", target("aes256-stored.zip"), "hello.txt", "lorem.txt"])

# 7-Zip split archive: .zip.001, .zip.002, ...
for f in os.listdir(out):
    if f.startswith("split7z.zip."):
        os.remove(os.path.join(out, f))
run([sevenzip, "a", "-tzip", "-mm=Copy", "-v1500b", os.path.join(out, "split7z.zip"), "lorem.txt", "random.bin"])

# Info-ZIP: ZipCrypto (data descriptors), streamed entries, symbolic link, split archive
run(["zip", "-q", "-X", "-P", "password", "-r", target("zipcrypto-infozip.zip")] + inputs)
with open(target("streamed-infozip.zip"), "wb") as f:
    p = subprocess.Popen(["zip", "-q", "-X", "-r", "-", "hello.txt", "lorem.txt", "random.bin", "dir"], cwd=src, stdout=subprocess.PIPE)
    f.write(p.stdout.read())
    p.wait()
with open(target("stdin-infozip.zip"), "wb") as f:
    subprocess.run(["zip", "-q", "-", "-"], input=lorem, stdout=f, check=True)
os.symlink("hello.txt", os.path.join(src, "link"))
run(["zip", "-q", "-y", target("symlink.zip"), "hello.txt", "link"])
os.remove(os.path.join(src, "link"))
big = os.path.join(work, "big")
os.makedirs(big)
with open(os.path.join(big, "big.bin"), "wb") as f:
    f.write(random.Random(3).randbytes(70000))
for f in ["split-infozip.zip", "split-infozip.z01"]:
    target(f)
run(["zip", "-q", "-0", "-s", "64k", os.path.join(out, "split-infozip.zip"), "big.bin"], cwd=big)
print("big.bin", hashlib.sha256(open(os.path.join(big, "big.bin"), "rb").read()).hexdigest())

# bsdtar
run(["bsdtar", "--format", "zip", "-cf", target("bsdtar.zip")] + inputs)
# bsdtar encrypts streamed entries: AE-1 (CRC) for the large files, AE-2 for the small ones, with data descriptors
run(["bsdtar", "--format", "zip", "--options", "zip:encryption=aes256", "--passphrase", "password", "-cf", target("aes-bsdtar.zip")] + inputs)
run(["bsdtar", "--format", "zip", "--options", "zip:encryption=zipcrypt", "--passphrase", "password", "-cf", target("zipcrypto-bsdtar.zip")] + inputs)

# Python: data descriptors (written to a non-seekable stream), zip64 descriptors, LZMA with an end marker, comments, UTF-8 names
class NonSeekable(io.RawIOBase):
    def __init__(self, f):
        self.f = f
    def writable(self):
        return True
    def write(self, b):
        return self.f.write(b)

def python_zip(name, entries, method=zipfile.ZIP_DEFLATED, seekable=True, force_zip64=False, comment=b""):
    with open(target(name), "wb") as raw:
        stream = raw if seekable else NonSeekable(raw)
        with zipfile.ZipFile(stream, "w", compression=method) as z:
            z.comment = comment
            for entry_name, data, entry_comment in entries:
                info = zipfile.ZipInfo(entry_name, date_time=(2024, 1, 2, 3, 4, 6))
                info.compress_type = method
                info.comment = entry_comment
                info.external_attr = 0o644 << 16
                with z.open(info, "w", force_zip64=force_zip64) as w:
                    w.write(data)

basic = [(n, d, b"") for n, d in files.items()]
python_zip("descriptor-python.zip", basic, seekable=False)
python_zip("descriptor-zip64-python.zip", basic, seekable=False, force_zip64=True)
python_zip("zip64-python.zip", basic, force_zip64=True)
python_zip("lzma-eos-python.zip", basic, method=zipfile.ZIP_LZMA)
python_zip("comments.zip", [("hello.txt", files["hello.txt"], "The entry comment".encode())], comment=b"The archive comment")
python_zip("utf8-names.zip", [("café/日本.txt", files["hello.txt"], b""), ("naïve.txt", lorem, b"")])
with zipfile.ZipFile(target("empty.zip"), "w"):
    pass

# Hand-written archives: methods Python does not write, CP437 names, Unicode path extra, timestamps, zip64 end of central directory
def dos(year, month, day, hour, minute, second):
    return (hour << 11) | (minute << 5) | (second // 2), ((year - 1980) << 9) | (month << 5) | day

def write_zip(name, entries, comment=b"", zip64_eocd=False, prefix=b""):
    """entries: dicts with name (bytes), data (compressed bytes), content (uncompressed), method, flags, extra, descriptor ('sig', 'nosig', 'sig64', 'nosig64' or None), zip64, made_by, external."""
    body = bytearray(prefix)
    central = bytearray()
    for e in entries:
        offset = len(body) - len(prefix)
        crc = zlib.crc32(e["content"]) & 0xFFFFFFFF
        csize, usize = len(e["data"]), len(e["content"])
        time, date = e.get("dos", dos(2024, 1, 2, 3, 4, 6))
        flags = e.get("flags", 0)
        descriptor = e.get("descriptor")
        zip64 = e.get("zip64", False)
        if descriptor:
            flags |= 8
        extra = e.get("extra", b"")
        local_extra = extra
        central_extra = extra
        if zip64:
            local_extra = struct.pack("<HHQQ", 1, 16, 0 if descriptor else usize, 0 if descriptor else csize) + extra
            central_extra = struct.pack("<HHQQQ", 1, 24, usize, csize, offset) + extra
        lcrc, lcsize, lusize = (0, 0, 0) if descriptor else (crc, csize, usize)
        if zip64:
            lcsize = lusize = 0xFFFFFFFF
        body += struct.pack("<IHHHHHIIIHH", 0x04034B50, 45 if zip64 else 20, flags, e["method"], time, date, lcrc, lcsize, lusize, len(e["name"]), len(local_extra)) + e["name"] + local_extra
        body += e["data"]
        if descriptor:
            fmt = "<IQQ" if descriptor.endswith("64") else "<III"
            d = struct.pack(fmt, crc, csize, usize)
            if descriptor.startswith("sig"):
                d = struct.pack("<I", 0x08074B50) + d
            body += d
        ccsize, cusize, coffset = (0xFFFFFFFF, 0xFFFFFFFF, 0xFFFFFFFF) if zip64 else (csize, usize, offset)
        central += struct.pack("<IHHHHHHIIIHHHHHII", 0x02014B50, e.get("made_by", 0x0314), 45 if zip64 else 20, flags, e["method"], time, date, crc, ccsize, cusize, len(e["name"]), len(central_extra), len(e.get("comment", b"")), 0, 0, e.get("external", 0o100644 << 16), coffset) + e["name"] + central_extra + e.get("comment", b"")
    cd_offset = len(body) - len(prefix)
    body += central
    count = len(entries)
    if zip64_eocd:
        eocd64_offset = len(body) - len(prefix)
        body += struct.pack("<IQHHIIQQQQ", 0x06064B50, 44, 45, 45, 0, 0, count, count, len(central), cd_offset)
        body += struct.pack("<IIQI", 0x07064B50, 0, eocd64_offset, 1)
        body += struct.pack("<IHHHHIIH", 0x06054B50, 0xFFFF, 0xFFFF, 0xFFFF, 0xFFFF, 0xFFFFFFFF, 0xFFFFFFFF, len(comment)) + comment
    else:
        body += struct.pack("<IHHHHIIH", 0x06054B50, 0, 0, count, count, len(central), cd_offset, len(comment)) + comment
    with open(target(name), "wb") as f:
        f.write(body)

def zstd(data):
    return subprocess.run(["zstd", "-q", "-c", "-19", "--no-check"], input=data, stdout=subprocess.PIPE, check=True).stdout

def plain(name, content, method, data, **kwargs):
    return dict(name=name.encode(), content=content, method=method, data=data, **kwargs)

names = ["hello.txt", "lorem.txt", "random.bin", "empty.txt", "dir/nested.txt"]
write_zip("zstd.zip", [plain(n, files[n], 93, zstd(files[n])) for n in names])
write_zip("xz.zip", [plain(n, files[n], 95, lzma.compress(files[n], format=lzma.FORMAT_XZ, check=lzma.CHECK_CRC64)) for n in names])
write_zip("zip64-eocd.zip", [plain(n, files[n], 0, files[n], zip64=True) for n in names], zip64_eocd=True, comment=b"zip64")
# tricky.bin contains a data descriptor signature with the size of the bytes before it (but not their CRC) and a header signature
tricky = b"abc" + b"PK\x07\x08" + struct.pack("<III", 0, 3, 3) + b"PK\x03\x04" + b"end"
print("tricky.bin", hashlib.sha256(tricky).hexdigest())
write_zip("descriptor-stored.zip", [plain(n, files[n], 0, files[n], descriptor=d) for n, d in zip(names, ["sig", "nosig", "sig", "nosig", "sig"])] + [plain("tricky.bin", tricky, 0, tricky, descriptor="sig")])
write_zip("descriptor-stored-zip64.zip", [plain(n, files[n], 0, files[n], descriptor="sig64", zip64=True) for n in names])
compressor = lambda d: (lambda c: c.compress(d) + c.flush())(zlib.compressobj(9, zlib.DEFLATED, -15))
write_zip("descriptor-deflate-nosig.zip", [plain(n, files[n], 8, compressor(files[n]), descriptor="nosig") for n in names])

# CP437 names, without the UTF-8 flag, from MS-DOS (backslash separators); and a Unicode path extra field (0x7075)
cp437_name = "résumé\\café.txt".encode("cp437")
unicode_name = "日本語.txt".encode("utf-8")
fallback = b"???.txt"
upath = struct.pack("<BI", 1, zlib.crc32(fallback) & 0xFFFFFFFF) + unicode_name
write_zip("cp437-names.zip", [
    dict(name=cp437_name, content=files["hello.txt"], method=0, data=files["hello.txt"], made_by=0x0014, external=0x20),
    dict(name=fallback, content=files["hello.txt"], method=0, data=files["hello.txt"], made_by=0x0314, extra=struct.pack("<HH", 0x7075, len(upath)) + upath),
])

# Timestamps: NTFS (0x000A) on the first entry, extended timestamp (0x5455) on the second, DOS only on the third
def filetime(seconds):
    return seconds * 10_000_000 + 116444736000000000
ntfs = struct.pack("<IHHQQQ", 0, 1, 24, filetime(1700000000), filetime(1700000100), filetime(1600000000))
ut = struct.pack("<BI", 1, 1650000000)
write_zip("timestamps.zip", [
    plain("ntfs.txt", b"ntfs", 0, b"ntfs", extra=struct.pack("<HH", 0x000A, len(ntfs)) + ntfs, made_by=0x000A, external=0x20),
    plain("unix.txt", b"unix", 0, b"unix", extra=struct.pack("<HH", 0x5455, len(ut)) + ut),
    plain("dos.txt", b"dos", 0, b"dos", dos=dos(2010, 6, 15, 13, 45, 30)),
])

# Data prepended to an archive (a self-extracting stub), without adjusting the offsets
with open(os.path.join(out, "deflate.zip"), "rb") as f:
    zipped = f.read()
with open(target("sfx.zip"), "wb") as f:
    f.write(b"MZ" + random.Random(4).randbytes(1022) + zipped)

shutil.rmtree(work)
```
